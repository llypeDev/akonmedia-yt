// {% translate "Your server, for the student signed in. Keep the secret out of the page." %}
function qqplay_token(array $student): string {
    $key = '{{ integration.key }}';
    $secret = getenv('QQPLAY_SECRET');
    $b64 = fn($data) => rtrim(strtr(base64_encode($data), '+/', '-_'), '=');
    $header = $b64(json_encode(['alg' => 'HS256', 'typ' => 'JWT', 'kid' => $key]));
    $claims = $b64(json_encode([
        'sub' => (string) $student['id'],
        'name' => $student['name'],
        'picture' => $student['photo_url'] ?? null,   // https, {% translate "optional" %}
        'moderator' => $student['is_teacher'] ?? false, // {% translate "may hide messages" %}
        'iat' => time(),
        'exp' => time() + 600,
    ]));
    $signature = $b64(hash_hmac('sha256', "$header.$claims", $secret, true));
    return "$header.$claims.$signature";
}

const crypto = require("crypto");

function qqplayToken(student) {
  const b64 = (value) => Buffer.from(JSON.stringify(value)).toString("base64url");
  const now = Math.floor(Date.now() / 1000);
  const header = b64({ alg: "HS256", typ: "JWT", kid: "{{ integration.key }}" });
  const claims = b64({
    sub: String(student.id),
    name: student.name,
    picture: student.photoUrl, // https, {% translate "optional" %}
    moderator: Boolean(student.isTeacher), // {% translate "may hide messages" %}
    iat: now,
    exp: now + 600,
  });
  const signature = crypto
    .createHmac("sha256", process.env.QQPLAY_SECRET)
    .update(`${header}.${claims}`)
    .digest("base64url");
  return `${header}.${claims}.${signature}`;
}

import base64, hashlib, hmac, json, os, time


def qqplay_token(student):
    def b64(data):
        return base64.urlsafe_b64encode(data).rstrip(b"=").decode()

    now = int(time.time())
    header = b64(json.dumps({"alg": "HS256", "typ": "JWT", "kid": "{{ integration.key }}"}).encode())
    claims = b64(json.dumps({
        "sub": str(student["id"]),
        "name": student["name"],
        "picture": student.get("photo_url"),  # https, {% translate "optional" %}
        "moderator": bool(student.get("is_teacher")),  # {% translate "may hide messages" %}
        "iat": now,
        "exp": now + 600,
    }).encode())
    secret = os.environ["QQPLAY_SECRET"].encode()
    signature = b64(hmac.new(secret, f"{header}.{claims}".encode(), hashlib.sha256).digest())
    return f"{header}.{claims}.{signature}"
